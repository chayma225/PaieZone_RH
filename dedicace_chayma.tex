\begin{center}
\textit{\Large DÉDICACE}
\end{center}

\vspace{0.2cm}

\begin{center}
\textit{C'est le c\oe ur débordant de gratitude que je dédie ce projet
à tous ceux qui ont illuminé mon chemin et contribué, de près ou de loin,
à faire de cette réussite une réalité.}
\end{center}

\vspace{0.4cm}

\begin{center}
\noindent\textbf{\textit{À mon Père,}}\\
\textit{pilier fondamental de ma vie, symbole de force et de persévérance.
Merci pour tes sacrifices, ton soutien indéfectible et les valeurs que tu m'as inculquées,
qui ont guidé chacun de mes pas. Tu es ma plus grande source d'inspiration.}
\end{center}

\vspace{0.3cm}

\begin{center}
\noindent\textbf{\textit{À ma Mère,}}\\
\textit{fontaine d'amour et de tendresse sans limites.
Merci pour tes prières, ton accompagnement constant et ta présence sans faille.
Tu es ma plus grande bénédiction.}
\end{center}

\vspace{0.3cm}

\begin{center}
\noindent\textbf{\textit{À mon grand frère Khalil,}}\\
\textit{mon idole et mon inspiration. Merci d'être toujours là pour moi, à chaque instant
où le doute s'installe, et de m'insuffler la force de continuer. Tu es mon plus beau modèle
de réussite et ma plus grande fierté.}
\end{center}

\vspace{0.3cm}

\begin{center}
\noindent\textbf{\textit{À mon frère Karim et ma petite s\oe ur Chadha,}}\\
\textit{pour leur présence chaleureuse et leur complicité sincère au quotidien.
Merci d'avoir apporté légèreté et équilibre à ma vie.}
\end{center}

\vspace{0.3cm}

\begin{center}
\noindent\textbf{\textit{À mon ami de toujours,}}\\
\textit{qui a cru en moi bien avant que je trouve ma voie et me pousse chaque jour
à donner le meilleur de moi-même. Merci d'être ce repère précieux vers la réalisation
de mes rêves.}
\end{center}

\vspace{0.3cm}

\begin{center}
\noindent\textbf{\textit{À mon bin\^ome et s\oe ur de c\oe ur Takwa,}}\\
\textit{ma partenaire de projet, de défis et de victoires. Merci pour ton amitié sincère,
ta fidélité et ton soutien à chaque étape. Partager ce parcours avec toi a été
un immense honneur.}
\end{center}

\vspace{0.3cm}

\begin{center}
\noindent\textbf{\textit{À mes amies Meryem et Wissal,}}\\
\textit{pour leur présence précieuse, leur sincérité et les souvenirs inoubliables.
Votre amitié a été une lumière tout au long de ce parcours.}
\end{center}

\vspace{0.4cm}

\begin{center}
\noindent\textbf{\textit{Enfin,}}\\
\textit{à toutes les personnes qui ont cru en moi et m'ont encouragée,
de près ou de loin, par leurs paroles, leurs gestes ou leur simple présence :
leur confiance a largement contribué à l'aboutissement de ce travail.}
\end{center}

\vspace{0.4cm}

\begin{flushright}
\textit{\Large Bouguerra Chayma}
\end{flushright}
